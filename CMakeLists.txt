cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/library.json" manifest)
string(JSON AECO_VERSION GET "${manifest}" version)
project(usdIfc VERSION ${AECO_VERSION} LANGUAGES CXX)
find_package(pxr CONFIG REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH TRUE)
set(CMAKE_BUILD_WITH_INSTALL_RPATH TRUE)
set(USDAECO_IFC_DEFAULT_PYTHON "python3" CACHE STRING "Default converter interpreter or launcher")
add_subdirectory(usdIfc)
